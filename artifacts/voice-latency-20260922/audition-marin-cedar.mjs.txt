import fs from 'node:fs';import WebSocket from 'ws';
const root='artifacts/voice-latency-20260922';
const key=fs.readFileSync('.dev.vars','utf8').split('\n').find(l=>/^OPENAI_API_KEY\s*=/.test(l)).split('=').slice(1).join('=').trim().replace(/^(["'])(.*)\1$/,'$2');
const lines=[['ru','Теперь всё сходится: осталось два отклика. А если кандидат сможет только в следующем месяце? Попробуем.'],['en','That leaves two applications to clarify. What if the candidate is only free next month? Let’s try it.']];
const reports=[];
for(const voice of ['marin','cedar'])await new Promise(resolve=>{
 const ws=new WebSocket('wss://api.openai.com/v1/realtime?model=gpt-realtime-2.1',{headers:{Authorization:'Bearer '+key}});let i=0,chunks=[],text='',started=false;const r={model:'gpt-realtime-2.1',voice,speed:'default, no playback-rate change',instructions:'Speak in an easy conversational voice, like talking to one person beside you. Relaxed, matter-of-fact, lightly curious. Natural phrasing and small pauses. No announcer delivery, forced cheer, exaggerated emphasis or breathy performance. Read the supplied line exactly without extra words.',clips:[]};const deadline=setTimeout(()=>ws.terminate(),60000);
 const send=e=>ws.send(JSON.stringify(e));const ask=()=>{chunks=[];text='';send({type:'conversation.item.create',item:{type:'message',role:'user',content:[{type:'input_text',text:'Read exactly in its original language: '+lines[i][1]}]}});send({type:'response.create'});};
 ws.on('open',()=>send({type:'session.update',session:{type:'realtime',model:'gpt-realtime-2.1',output_modalities:['audio'],reasoning:{effort:'low'},audio:{input:{turn_detection:null},output:{format:{type:'audio/pcm',rate:24000},voice}},instructions:r.instructions}}));
 ws.on('message',raw=>{const e=JSON.parse(raw);if(e.type==='session.updated'&&!started){started=true;ask();}if(e.type==='response.output_audio.delta')chunks.push(Buffer.from(e.delta,'base64'));if(e.type==='response.output_audio_transcript.delta')text+=e.delta;if(e.type==='response.done'){const pcm=Buffer.concat(chunks),h=Buffer.alloc(44);h.write('RIFF');h.writeUInt32LE(36+pcm.length,4);h.write('WAVEfmt ',8);h.writeUInt32LE(16,16);h.writeUInt16LE(1,20);h.writeUInt16LE(1,22);h.writeUInt32LE(24000,24);h.writeUInt32LE(48000,28);h.writeUInt16LE(2,32);h.writeUInt16LE(16,34);h.write('data',36);h.writeUInt32LE(pcm.length,40);const name=`audition-${voice}-${lines[i][0]}.wav`;fs.writeFileSync(root+'/'+name,Buffer.concat([h,pcm]));r.clips.push({language:lines[i][0],requested:lines[i][1],transcript:text,seconds:pcm.length/48000,status:e.response.status,file:name});if(++i<lines.length)ask();else ws.close();}if(e.type==='error'){r.error=e.error?.code;ws.close();}});ws.on('close',()=>{clearTimeout(deadline);r.closed=true;reports.push(r);fs.writeFileSync(root+'/voice-auditions.json',JSON.stringify(reports,null,2)+'\n');resolve();});ws.on('error',()=>{r.error='connection';});
});console.log(JSON.stringify(reports));
