'use client';
import {useEffect,useRef,useState} from 'react';
import {createLiveVoice} from '@/lib/live-pcm-client';
import type {DemoContext} from '@/lib/live-demo';
import type {GameContext} from '@/lib/live-game';
type Entry={event:string;ms:number;[key:string]:unknown};
export default function Lab(){
 const [journal,setJournal]=useState<Entry[]>([]),[status,setStatus]=useState('idle'),[caption,setCaption]=useState('');
 const rows=useRef<Entry[]>([]),clock=useRef(0),voice=useRef<ReturnType<typeof createLiveVoice>|null>(null),ctx=useRef<AudioContext|null>(null),input=useRef<MediaStreamAudioDestinationNode|null>(null),buffers=useRef<AudioBuffer[]>([]),playing=useRef(false),source=useRef<AudioBufferSourceNode|null>(null),test=useRef(''),seq=useRef(0),lastInput=useRef(0),firstPacket=useRef(false),firstPlayed=useRef(false),done=useRef(false),timer=useRef<ReturnType<typeof setTimeout>|undefined>(undefined),draft=useRef<DemoContext>({kind:'hiring',phase:'case',choice:null,revision:0,language:'ru'});
 const feeding=useRef(false),advanced=useRef(false),captionRef=useRef(''),auto=useRef(false),quizAuto=useRef(false);
 const log=(event:string,data:Record<string,unknown>={})=>{const row={event,ms:Math.round((performance.now()-clock.current)*10)/10,test:test.current,...data};rows.current=[...rows.current,row];setJournal([...rows.current]);sessionStorage.setItem('navi-latency-'+location.search,JSON.stringify(rows.current));};
 function finish(){clearTimeout(timer.current);source.current?.stop();voice.current?.stop();log('stop.requested');}
 useEffect(()=>{const stored=sessionStorage.getItem('navi-latency-'+location.search);if(stored){rows.current=JSON.parse(stored);setJournal([...rows.current]);}},[]);
 useEffect(()=>()=>{clearTimeout(timer.current);source.current?.stop();voice.current?.dispose();},[]);
 async function start(kind:'hiring'|'quiz',language:'ru'|'en'='ru'){
  if(voice.current)return;
  auto.current=true;quizAuto.current=kind==='quiz';clock.current=performance.now();test.current='cold';rows.current=[];seq.current=0;log('explicit.start',{kind,wallClock:new Date().toISOString()});
  const context=new AudioContext({sampleRate:24000});ctx.current=context;input.current=context.createMediaStreamDestination();
  const env={context:()=>context,node:(c:AudioContext)=>{const n=new AudioWorkletNode(c,'navi-pcm',{numberOfInputs:1,numberOfOutputs:1,outputChannelCount:[1]});n.port.addEventListener('message',({data})=>{if(data.type==='level'&&data.inputLevel>.009)lastInput.current=performance.now()-clock.current;if(data.type==='playback'){playing.current=data.busy;log('playback',{busy:data.busy});if(!data.busy)setTimeout(advance,0);}if(data.type==='truncated')log('truncate',{playedMs:data.audio_end_ms});});return n;},microphone:async()=>{log('synthetic.microphone');return input.current!.stream;},socket:()=>{const s=new WebSocket('ws://127.0.0.1:5174/api/live/stream');s.addEventListener('message',({data})=>{const e=JSON.parse(data);if(e.type==='response.started'){firstPacket.current=false;firstPlayed.current=false;done.current=false;advanced.current=false;log(e.type,{revision:e.demo_revision});}else if(e.type==='session.output_audio.delta'&&!firstPacket.current){firstPacket.current=true;log('first.packet');}else if(['response.requested','input.speech_started','input.speech_stopped','audio.clear','session.started','session.closed','response.done','error'].includes(e.type)){log(e.type,{reason:e.reason,status:e.status,lastInputSignalMs:lastInput.current});if(e.type==='response.done'){done.current=true;setTimeout(advance,0);}}});s.addEventListener('close',()=>log('socket.closed'));return s;},fetch:(...a:Parameters<typeof fetch>)=>fetch(...a)};
  voice.current=createLiveVoice({onState:s=>setStatus(s.phase),onDiagnostic:(event,details)=>log(event,details),onSpeech:f=>{if(f.speaking&&!firstPlayed.current){firstPlayed.current=true;log('first.played',{level:f.level,audioContext:context.state,baseLatency:context.baseLatency,outputLatency:context.outputLatency});if(auto.current&&test.current==='help')setTimeout(()=>{play(0);test.current='barge';log('barge.start');},200);else if(auto.current&&test.current==='barge')setTimeout(change,200);}},onCaption:text=>{captionRef.current=text;setCaption(text);}},env);
  if(kind==='hiring'){draft.current={...draft.current,language};voice.current.updateDemo(draft.current);await voice.current.start(language);}else{voice.current.updateGame({phase:'welcome',step:0,answers:[],selected:null,interest:''});await voice.current.start('ru');}
  buffers.current=await Promise.all([language,language].map(async f=>context.decodeAudioData(await(await fetch('/voice-latency-lab/'+f+'.wav')).arrayBuffer())));
  timer.current=setTimeout(()=>{log('bounded.timeout');finish();},150000);
 }
 function play(i:number){
  if(!ctx.current||!buffers.current[i]||!voice.current)return;
  if(i===1){draft.current={...draft.current,phase:'result',choice:'week',revision:1};voice.current.updateDemo(draft.current);}
  feeding.current=true;test.current='spoken_'+(i+1);lastInput.current=0;log('input.start',{duration:buffers.current[i].duration});
  const s=ctx.current.createBufferSource();source.current=s;s.buffer=buffers.current[i];s.connect(input.current!);s.onended=()=>{feeding.current=false;log('input.file_ended',{lastInputSignalMs:lastInput.current});};s.start();
 }
 function change(){firstPlayed.current=false;test.current='context';draft.current={...draft.current,phase:'result',choice:'month',revision:2};log('context.change');voice.current?.updateDemo(draft.current);}
 function help(){test.current='help';log('help.tap');voice.current?.greet();}
 function quiz(){const step=seq.current++;test.current=step<5?'quiz_'+step:'quiz_result';log('context.change');const game:GameContext={phase:step<5?'quiz':'result',step:Math.min(step,4),answers:Array(Math.min(step,5)).fill(0),selected:step<5?null:0,interest:''};voice.current?.updateGame(game);}
 function advance(){if(!auto.current||advanced.current||!done.current||playing.current||feeding.current||!firstPlayed.current)return;advanced.current=true;log('turn.finished',{caption:captionRef.current});if(quizAuto.current){if(seq.current<=5)quiz();else{auto.current=false;finish();log('rehearsal.complete');}return;}if(test.current==='cold')play(0);else if(test.current==='spoken_1')play(1);else if(test.current==='spoken_2')help();else if(test.current==='context'){auto.current=false;finish();log('rehearsal.complete');}}
 function snapshot(){log('snapshot',{caption,status,playing:playing.current,done:done.current});}
 return <main style={{padding:24,maxWidth:1100}}><h1>Synthetic voice latency rehearsal</h1><p>No device microphone, contacts or database. Real provider and real browser output worklet; explicit test only.</p><p role="status">{status}</p><div style={{display:'flex',gap:12,flexWrap:'wrap'}}><button onClick={()=>void start('hiring')}>Start Russian demo test</button><button onClick={()=>void start('hiring','en')}>Start English demo test</button><button onClick={()=>void start('quiz')}>Start Russian quiz test</button><button onClick={()=>play(0)}>Play same question 1</button><button onClick={()=>play(1)}>Choose week and play question 2</button><button onClick={change}>Change to month</button><button onClick={help}>Explicit help</button><button onClick={quiz}>Next quiz screen</button><button onClick={snapshot}>Capture transcript</button><button onClick={finish}>Stop test</button><button onClick={()=>{const a=document.createElement('a');a.href=URL.createObjectURL(new Blob([JSON.stringify(rows.current,null,2)],{type:'application/json'}));a.download='navi-latency-trace.json';a.click();}}>Download trace</button></div><p>{caption}</p><output data-test-state>{test.current}:{playing.current?'playing':done.current?'finished':'waiting'}</output><pre data-journal style={{fontSize:11,whiteSpace:'pre-wrap'}}>{JSON.stringify(journal,null,2)}</pre></main>
}
